### 2. Real-Time Animation (useFrame Magic)
Animate any prop (`scale`, `offset`, `alpha`, `colorA/B`) inside `useFrame` for fluid motion. Combine with mouse scroll for parallax.

```tsx
import { useRef } from 'react';
import { useFrame } from '@react-three/fiber';
import { Noise } from 'lamina';

const AnimatedNoise = () => {
  const noiseRef = useRef();
  useFrame((state) => {
    noiseRef.current.scale = 1 + Math.sin(state.clock.elapsedTime) * 0.5;
    noiseRef.current.offset = [state.clock.elapsedTime * 0.1, 0, 0];
  });
  return <Noise ref={noiseRef} type="perlin" scale={2} colorA="#ff5500" colorB="#00ffff" alpha={0.7} />;
```
