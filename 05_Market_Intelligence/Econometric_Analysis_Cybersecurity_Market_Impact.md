## 3.2 Cyber Risk Quantification Frameworks

### FAIR (Factor Analysis of Information Risk)

```python
# Aplicação FAIR ao caso Petrobras

class FAIR_Analysis:
    """
    Factor Analysis of Information Risk
    """
    
    def __init__(self):
        # Threat Event Frequency
        self.TEF = 0.5  # 1 evento a cada 2 anos (estimado)
        
        # Vulnerability
        self.TC = 0.7   # Threat Capability (Everest = high)
        self.CS = 0.4   # Control Strength (pre-breach)
        
    def calculate_loss_event_frequency(self):
        """LEF = TEF × Vulnerability"""
        vulnerability = self.TC - self.CS  # 0.3
        if vulnerability < 0:
            vulnerability = 0.05  # Minimum residual
        
        LEF = self.TEF * vulnerability
        return LEF  # 0.15 (15% chance anual)
    
    def calculate_loss_magnitude(self):
        """PLM - Primary Loss Magnitude"""
        primary_losses = {
            'productivity': 2000000,      # Downtime estimado
            'response': 5000000,          # Forensics, legal
            'replacement': 1000000,       # Systems recovery
            'fines_judgements': 15000000, # ANPD + class action
            'competitive_advantage': 50000000,  # Seismic data value
            'reputation': 20000000        # Brand damage
        }
        
        # SLM - Secondary Loss Magnitude
        secondary_losses = {
            'stock_price_impact': 1500000000,  # 1.2% de R$ 125B
            'customer_churn': 0,               # B2B - baixo churn
            'increased_cost_capital': 50000000,
            'insurance_premium_increase': 15000000
        }
        
        total_plm = sum(primary_losses.values())
        total_slm = sum(secondary_losses.values()) * 0.1  # 10% probability
        
        return {
            'primary': total_plm,      # US$ 93M
            'secondary': total_slm,     # US$ 157M
            'total_expected': total_plm + total_slm  # US$ 250M
        }
    
    def calculate_risk(self):
        """Risk = LEF × LM"""
        lef = self.calculate_loss_event_frequency()
        lm = self.calculate_loss_magnitude()
        
        annual_risk = lef * lm['total_expected']
        
        return {
            'annualized_risk': annual_risk,  # US$ 37.5M/year
            'lef': lef,
            'loss_magnitude': lm,
            'risk_rating': 'HIGH' if annual_risk > 20e6 else 'MEDIUM'
        }

# Resultado para Petrobras (pre-Nov 2025):
# Annualized Cyber Risk: US$ 37.5M
# Post-breach (controls melhorados): US$ 15-20M
```

### Modelagem de Risco de Vendor

```python
class SupplyChainRiskModel:
    """
    Modelagem econômica de risco de supply chain
    """
    
    def __init__(self, company_profile):
        self.vendors = company_profile['vendors']
        self.revenue = company_profile['annual_revenue']
        self.industry = company_profile['industry']
        
    def calculate_vendor_portfolio_risk(self):
        """
        Cálculo de risco agregado do portfolio de vendors
        """
        total_risk = 0
        vendor_risks = []
        
        for vendor in self.vendors:
            # Probabilidade de comprometimento
            pd = self._vendor_default_probability(vendor)
            
            # Exposure at Default (valor em risco)
            ead = vendor['annual_spend'] + vendor['contract_value'] * 0.3
            
            # Loss Given Default
            lgd = self._industry_lgd(self.industry, vendor['criticality'])
            
            # Expected Loss
            el = pd * ead * lgd
            
            # Correlation com outros vendors
            correlation_factor = self._correlation_adjustment(vendor)
            
            risk_contribution = el * correlation_factor
            total_risk += risk_contribution
            
            vendor_risks.append({
                'vendor': vendor['name'],
                'pd': pd,
                'ead': ead,
                'lgd': lgd,
                'expected_loss': el,
                'risk_contribution': risk_contribution
            })
        
        # Value at Risk (Monte Carlo)
        var_95 = self._monte_carlo_var(vendor_risks, confidence=0.95)
        var_99 = self._monte_carlo_var(vendor_risks, confidence=0.99)
        
        return {
            'portfolio_expected_loss': total_risk,
            'var_95': var_95,
            'var_99': var_99,
            'as_percentage_of_revenue': total_risk / self.revenue,
            'vendor_breakdown': vendor_risks
        }
    
    def _vendor_default_probability(self, vendor):
        """
        Probabilidade de breach/comprometimento do vendor
        """
        base_pd = 0.03  # 3% base anual
        
        # Ajustes
        if vendor['security_rating'] == 'LOW':
            base_pd *= 2.5
        elif vendor['security_rating'] == 'MEDIUM':
            base_pd *= 1.5
            
        if vendor['has_access_to_ot']:
            base_pd *= 1.3
            
        if vendor['international']:
            base_pd *= 1.2
            
        return min(base_pd, 0.20)  # Cap em 20%
    
    def _industry_lgd(self, industry, criticality):
        """Loss Given Default por indústria"""
        lgd_matrix = {
            'energy': {
                'CRITICAL': 0.45,
                'HIGH': 0.30,
                'MEDIUM': 0.15,
                'LOW': 0.08
            },
            'finance': {
                'CRITICAL': 0.50,
                'HIGH': 0.35,
                'MEDIUM': 0.20,
                'LOW': 0.10
            }
        }
        return lgd_matrix.get(industry, {}).get(criticality, 0.25)
```
