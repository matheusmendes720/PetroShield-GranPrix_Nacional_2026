# Vendor Risk Economic Model

class SupplyChainVaR:
    """
    Value at Risk para supply chain
    """
    
    def __init__(self):
        self.vendors = load_vendor_portfolio()  # 10,000 vendors
        self.scenarios = ['base', 'stress', 'extreme']
        
    def calculate_portfolio_risk(self):
        """
        Cálculo VaR do portfolio de vendors
        """
        total_exposure = 0
        
        for vendor in self.vendors:
            # Exposure at Default
            EAD = vendor.annual_spend + vendor.contract_value
            
            # Probability of Default (breach/disruption)
            PD = vendor.risk_score / 100
            
            # Loss Given Default
            if vendor.criticality == 'CRITICAL':
                LGD = 0.40  # 40% loss
            elif vendor.criticality == 'HIGH':
                LGD = 0.25
            else:
                LGD = 0.10
            
            # Expected Loss
            EL = EAD * PD * LGD
            total_exposure += EL
        
        # VaR 95% e 99%
        var_95 = np.percentile(simulated_losses, 95)
        var_99 = np.percentile(simulated_losses, 99)
        
        return {
            'expected_annual_loss': total_exposure,  # ~US$ 150-200M
            'var_95': var_95,  # ~US$ 500M
            'var_99': var_99   # ~US$ 1.2B
